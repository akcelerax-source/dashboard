// ==========================================================================
// NODEX - ACE on-board task allocation (System 3 only). One instance per robot.
//
// Contract-Net with a real bid window and a joint award (WS3, r6):
//   announce  : the task board (or a peer, for orphaned work) broadcasts the
//               open tasks; every robot receives them in the same order.
//   bid       : each eligible robot answers ONE message per announcement
//               round with its predicted cost for the first tasks of the round
//               (bundle), plus its local estimate of tasks in flight.
//   window    : the award happens one tick after the round was received, when
//               every robot's bid has been delivered (agents tick in turn and
//               the bus delivers synchronously, so a bid sent in tick t reaches
//               every peer by its tick t+1).
//   award     : every bidder solves the SAME min-cost one-to-one assignment
//               over the bids it received (first K tasks of the round in board
//               order x bidders, K = free admission slots) and claims only the
//               task assigned to itself. No robot decides for another one and
//               no central solver exists: identical inputs give identical
//               outputs on every robot. The task board still accepts only the
//               first claim of an open task, so lost bids cannot duplicate work.
//
// Bid cost (ws3-eta-bid): predicted time to reach the pickup, in seconds:
//   graph route length on the robot's own map / cruise speed
//   + ACE risk term, - battery incentive (the legacy weights, in seconds).
//   A congestion term (peer queues / planned flow / own per-node wait EMA)
//   was tested and removed: no measurable benefit (docs/changes/ws3.md).
// Without ws3-eta-bid the legacy straight-line cost is used.
// ==========================================================================

import { MESSAGE_TYPES } from "../decentralized/PeerCommunicationBus.js";
import { MapGeometryEngine } from "../map-geometry.js";
import { concurrentTaskLimit } from "../admission-control.js";
import { needsService } from "../task-lifecycle.js";
import { hungarian } from "../centralized/HungarianAssignment.js";
import { aceFeature } from "./ace-features.js";

/** Nominal cruise speed in px per simulated second (targetVelocity 1.2 x 22 px). */
export const CRUISE_PX_S = 1.2 * 22;
// Extra tasks bid on beyond the free admission slots (covers estimate skew).
const BID_MARGIN = 2;
// A bid belongs to a round only when it was made within this long of the
// robot's own receipt of the announcement (one tick of delivery skew).
const ROUND_SKEW_S = 0.25;
// ws3-busy-bid: a robot on the loaded leg that will be free within this many
// seconds bids too (cost = its remaining time + the leg from its drop). It
// never claims while busy: winning only keeps the task for it (every robot
// computes the same award, so nobody else claims it) until it is free.
const BUSY_HORIZON_S = 12;
// ws3-task-swap: a holder still far from its pickup offers the task once
// (re-announcement with its own ETA as reserve); an idle peer takes it over
// only when it is faster by the hysteresis margin. One handover per task.
const SWAP_CHECK_S = 1.0;
const SWAP_MIN_ETA_S = 6;
const SWAP_MARGIN_S = 3;
const SWAP_MARGIN_FRAC = 0.2;
const isFailed = (st) => st === "ERROR" || st === "error" || st === "failed";

function hashIds(ids) {
  let h = 5381;
  for (const id of ids) for (let i = 0; i < id.length; i++) h = ((h << 5) + h + id.charCodeAt(i)) | 0;
  return (h >>> 0).toString(36);
}

export class AceTaskAllocator {
  constructor(agent) {
    this.agent = agent;
    this.tickNo = 0;
    this.pending = [];          // [{ sender, task }] announcements received since the last tick
    this.rounds = new Map();    // key -> { key, sender, tasks, ids, seenTick, seenAt, bids: Map robotId -> { costs: Map, act } }
    this.early = new Map();     // key -> [{ sender, p }] bids that arrived before their announcement
    this._pathCache = new Map();
    this._swapOffered = new Set();
    this._swapCheckAt = -Infinity;
    this.stats = { rounds: 0, bidsSent: 0, awardsWon: 0, claimsRefused: 0, awardComputations: 0 };
  }

  get ls() { return this.agent.localState; }

  // ── inbox ────────────────────────────────────────────────────────────────

  /** A task announcement (board or peer re-announcement) was received. */
  onAnnouncement(task, sender = "SYSTEM", payload = null) {
    if (!task || !task.id || this.ls.status === "ERROR") return;
    const swap = payload && payload.swapFrom ? { from: payload.swapFrom, reserve: payload.reserve } : null;
    this.pending.push({ sender, task, swap });
  }

  /** A peer's bundled bid was received. */
  onBid(sender, p) {
    if (!p || !p.round || !Array.isArray(p.costs)) return;
    const r = this.rounds.get(p.round);
    if (!r) {
      const list = this.early.get(p.round) || [];
      list.push({ sender, p });
      this.early.set(p.round, list);
      return;
    }
    this._addBid(r, sender, p);
  }

  _addBid(r, sender, p) {
    if (typeof p.at === "number" && Math.abs(p.at - r.seenAt) > ROUND_SKEW_S) return; // stale round
    r.bids.set(sender, { costs: new Map(p.costs), act: Number.isFinite(p.act) ? p.act : 0 });
  }

  // ── per tick (after the inbox) ───────────────────────────────────────────

  tick(taskRegistry) {
    this.tickNo++;
    this._checkLease(taskRegistry);
    this._award(taskRegistry);
    this._maybeOfferSwap(taskRegistry);
    this._openRounds();
    this._bid();
    this.early.clear();
  }

  /** Groups this tick's announcements into rounds (one per sender). */
  _openRounds() {
    if (!this.pending.length) return;
    const now = this.agent.simNow;
    const bySender = new Map();
    for (const { sender, task, swap } of this.pending) {
      if (swap) {
        // Handover offer: its own one-task round.
        const key = `${sender}:swap:${task.id}`;
        const r = { key, sender, tasks: [task], ids: [task.id], seenTick: this.tickNo, seenAt: now, bids: new Map(), bidDone: false, swap };
        this.rounds.set(key, r);
        for (const e of this.early.get(key) || []) this._addBid(r, e.sender, e.p);
        continue;
      }
      if (!bySender.has(sender)) bySender.set(sender, []);
      const list = bySender.get(sender);
      if (!list.some(t => t.id === task.id)) list.push(task);
    }
    this.pending = [];
    for (const [sender, tasks] of bySender) {
      // Tasks keep the announcer's order (the board announces its own queue
      // order), identical on every robot.
      const ids = tasks.map(t => t.id);
      const key = `${sender}:${hashIds(ids)}:${ids.length}`;
      const r = { key, sender, tasks, ids, seenTick: this.tickNo, seenAt: now, bids: new Map(), bidDone: false };
      this.rounds.set(key, r);
      this.stats.rounds++;
      for (const e of this.early.get(key) || []) this._addBid(r, e.sender, e.p);
    }
  }

  /** Same eligibility as the legacy Contract-Net bid (RobotAgent.handleTaskAnnouncement). */
  eligible() {
    const ls = this.ls;
    if (ls.status === "ERROR") return false;
    if (ls.currentTaskId) return this._busyRemaining() !== null ? "busy" : false;
    const returning = ls.parkingBay && !ls._maneuver && (ls.parkingBay.kind === "HOME" || ls.parkingBay.kind === "STANDBY");
    if ((ls.status !== "IDLE" && !returning) || ls.currentTaskId) return false;
    if (ls.hitlHold || ls.serviceState || ls.handling || needsService(ls)) return false;
    if (ls.battery < 20) return false;
    return MapGeometryEngine.bayExitClear(ls, this.agent.peerCache.values());
  }

  /**
   * ws3-busy-bid (fleets without an admission cap only: under a cap a kept
   * task would hold a slot): seconds until I am free, when I am on the loaded
   * leg and will finish within BUSY_HORIZON_S; otherwise null.
   */
  _busyRemaining() {
    const ls = this.ls;
    if (!aceFeature("ws3-busy-bid") || Number.isFinite(concurrentTaskLimit(this.agent.fleetSize))) return null;
    if (ls.pickedUp !== true || !ls.currentGoal || ls.hitlHold || ls._maneuver || ls.battery < 30) return null;
    // A robot held up in traffic is not "finishing soon": its route length
    // no longer predicts when it is free, so it stops bidding while busy.
    if (!ls.handling && (ls.status === "WAITING" || (ls.stalledDuration || 0) > 1.0)) return null;
    let rem;
    if (ls.handling) rem = ls.handling.remaining;
    else rem = this._route(ls.currentGoal).len / CRUISE_PX_S + 1.0; // + unload dwell
    if (rem > BUSY_HORIZON_S) return null;
    // Never keep work for longer than the horizon it was promised for.
    const now = this.agent.simNow;
    if (!this._busySince || this._busySince.task !== ls.currentTaskId) this._busySince = { task: ls.currentTaskId, t: now };
    return now - this._busySince.t > BUSY_HORIZON_S ? null : rem;
  }

  /** Free admission slots as seen from my own state and my peers' broadcasts. */
  _freeSlots() {
    const cap = concurrentTaskLimit(this.agent.fleetSize);
    return Number.isFinite(cap) ? Math.max(0, cap - this.agent._localActiveTaskEstimate()) : Infinity;
  }

  _bid() {
    const fresh = [...this.rounds.values()].filter(r => r.seenTick === this.tickNo && !r.bidDone);
    if (!fresh.length) return;
    const ok = this.eligible();
    const act = this.agent._localActiveTaskEstimate();
    const limit = Math.min(this._freeSlots(), Math.max(1, this.agent.fleetSize || 1)) + BID_MARGIN;
    for (const r of fresh) {
      r.bidDone = true;
      let costs = [];
      if (r.swap) {
        // Handover round: the holder bids its reserve, idle robots their ETA.
        if (r.swap.from === this.agent.robotId) costs = [[r.ids[0], r.swap.reserve]];
        else if (ok === true && r.tasks[0].pickup) {
          const c = this.bidCost(r.tasks[0]);
          if (Number.isFinite(c)) costs = [[r.ids[0], Math.round(c * 1000) / 1000]];
        }
        if (!costs.length) continue;
        r.bids.set(this.agent.robotId, { costs: new Map(costs), act });
        this.agent.peerBus.broadcast(this.agent.robotId, MESSAGE_TYPES.TASK_BID, { round: r.key, at: r.seenAt, act, costs });
        this.stats.bidsSent++;
        continue;
      }
      if (!ok) continue;
      for (const task of r.tasks.slice(0, limit)) {
        if (!task.pickup) continue;
        const c = this.bidCost(task);
        if (Number.isFinite(c)) costs.push([task.id, Math.round(c * 1000) / 1000]);
      }
      if (!costs.length) continue;
      const payload = { round: r.key, at: r.seenAt, act, costs };
      r.bids.set(this.agent.robotId, { costs: new Map(costs), act });
      this.agent.peerBus.broadcast(this.agent.robotId, MESSAGE_TYPES.TASK_BID, payload);
      this.stats.bidsSent++;
      this.agent.logDecision("SUBMIT_BID", `Bid on ${costs.length} task(s) of round ${r.key} (best ${Math.min(...costs.map(c => c[1])).toFixed(1)}).`, { round: r.key });
    }
  }

  /** Joint award of every round whose bid window has closed. */
  _award(taskRegistry) {
    const me = this.agent.robotId;
    const taken = new Set(); // robots assigned by an earlier round this tick (same on every robot)
    for (const [key, r] of this.rounds) {
      if (r.seenTick >= this.tickNo) continue;
      this.rounds.delete(key);
      if (!r.bids.has(me)) continue; // I did not bid: I cannot be awarded anything
      this.stats.awardComputations++;
      const bidders = [...r.bids.keys()].filter(id => !taken.has(id)).sort();
      let act = 0;
      for (const b of r.bids.values()) act = Math.max(act, b.act);
      const cap = concurrentTaskLimit(this.agent.fleetSize);
      // A handover moves an active task: it needs no free admission slot.
      const free = r.swap ? 1 : Number.isFinite(cap) ? Math.max(0, cap - act) : Infinity;
      const K = Math.min(r.ids.length, free, bidders.length);
      if (K <= 0) continue;
      const cols = r.tasks.slice(0, K);
      // Only the K best bidders of each task can appear in an optimal
      // assignment of K tasks; restrict the rows to them (exact, smaller).
      const rowSet = new Set();
      for (const t of cols) {
        const ranked = bidders.filter(id => Number.isFinite(r.bids.get(id).costs.get(t.id)))
          .sort((a, b) => r.bids.get(a).costs.get(t.id) - r.bids.get(b).costs.get(t.id) || (a < b ? -1 : 1));
        for (const id of ranked.slice(0, K)) rowSet.add(id);
      }
      const rows = [...rowSet].sort();
      if (!rows.length) continue;
      const matrix = rows.map(id => cols.map(t => {
        const c = r.bids.get(id).costs.get(t.id);
        return Number.isFinite(c) ? c : Infinity;
      }));
      const pairs = hungarian(matrix);
      for (const [i] of pairs) taken.add(rows[i]);
      const mine = pairs.find(([i]) => rows[i] === me);
      if (!mine) continue;
      const task = cols[mine[1]];
      const ls = this.ls;
      if (r.swap) {
        // I won a handover round. The holder keeps it when it won itself;
        // otherwise I take the lease over on the board (the holder releases
        // when it sees that, see _checkLease).
        if (r.swap.from === me || ls.currentTaskId || this.eligible() !== true) continue;
        if (this.agent.claimTaskOwnership(task, taskRegistry, { handoverFrom: r.swap.from })) {
          this.stats.handoversTaken = (this.stats.handoversTaken || 0) + 1;
          this.agent.logDecision("TASK_HANDOVER_IN", `Took over ${task.id} from ${r.swap.from} (ETA ${matrix[mine[0]][mine[1]].toFixed(1)} s vs reserve ${r.swap.reserve.toFixed(1)} s).`);
        }
        continue;
      }
      if (ls.currentTaskId) {
        // Busy winner: the task stays open for me; I claim it once free.
        this.stats.keptForLater = (this.stats.keptForLater || 0) + 1;
        continue;
      }
      if (this.eligible() !== true) continue;
      const nBids = [...r.bids.values()].filter(b => Number.isFinite(b.costs.get(task.id))).length;
      if (this.agent.claimTaskOwnership(task, taskRegistry)) {
        this.stats.awardsWon++;
        const t = taskRegistry && taskRegistry.getTaskById(task.id);
        if (t) t.bidRound = { bids: nBids, winningCost: matrix[mine[0]][mine[1]], joint: rows.length + "x" + cols.length };
      } else {
        this.stats.claimsRefused++;
      }
    }
  }

  // ── task handover (ws3-task-swap) ────────────────────────────────────────

  /**
   * Lease check: my task, not loaded yet, is held by another robot on the task
   * board (handed over, or re-awarded while peers could not hear me): stop
   * working on it so the work is never done twice.
   */
  _checkLease(taskRegistry) {
    if (!aceFeature("ws3-task-swap") || !taskRegistry) return;
    const ls = this.ls;
    if (!ls.currentTaskId || ls.pickedUp !== false || ls.handling) return;
    const t = taskRegistry.getTaskById(ls.currentTaskId);
    if (!t || !t.assignedRobot || t.assignedRobot === this.agent.robotId) return;
    if (t.status !== "ASSIGNED" && t.status !== "EXECUTING") return;
    this.agent.logDecision("TASK_HANDOVER_OUT", `Task ${t.id} is now held by ${t.assignedRobot}; releasing it before pickup.`);
    this.stats.handoversGiven = (this.stats.handoversGiven || 0) + 1;
    this.agent._clearTask();
    this.agent._claimChanged = true;
  }

  /** Holder side: offer my task once when a free peer could plausibly reach its pickup faster. */
  _maybeOfferSwap(taskRegistry) {
    if (!aceFeature("ws3-task-swap") || !taskRegistry) return;
    const ls = this.ls;
    if (!ls.currentTaskId || ls.pickedUp !== false || !ls.currentPickup || ls.handling || ls._maneuver || ls.hitlHold) return;
    const now = this.agent.simNow;
    if (now - this._swapCheckAt < SWAP_CHECK_S) return;
    this._swapCheckAt = now;
    if (this._swapOffered.has(ls.currentTaskId)) return;
    const task = taskRegistry.getTaskById(ls.currentTaskId);
    if (!task || task.assignedRobot !== this.agent.robotId || (task.handovers || 0) > 0) return;
    const own = this._holderCost(ls.currentPickup);
    const eta = this._route(ls.currentPickup).len / CRUISE_PX_S;
    if (eta < SWAP_MIN_ETA_S) return;
    const reserve = own - Math.max(SWAP_MARGIN_S, SWAP_MARGIN_FRAC * eta);
    // Worth a round only if an idle peer is within straight-line reach.
    const reach = (eta - Math.max(SWAP_MARGIN_S, SWAP_MARGIN_FRAC * eta)) * CRUISE_PX_S;
    const fresh = this.agent.now() - 1500;
    let candidate = false;
    for (const p of this.agent.peerCache.values()) {
      if ((p.lastSeen ?? 0) < fresh || p.currentTaskId || isFailed(p.status) || typeof p.x !== "number") continue;
      if (Math.hypot(p.x - ls.currentPickup.x, p.y - ls.currentPickup.y) < reach) { candidate = true; break; }
    }
    if (!candidate) return;
    this._swapOffered.add(task.id);
    const payload = { task, swapFrom: this.agent.robotId, reserve: Math.round(reserve * 1000) / 1000 };
    this.agent.peerBus.broadcast(this.agent.robotId, MESSAGE_TYPES.TASK_ANNOUNCEMENT, payload);
    this.onAnnouncement(task, this.agent.robotId, payload);
    this.stats.swapOffers = (this.stats.swapOffers || 0) + 1;
    this.agent.logDecision("TASK_SWAP_OFFER", `Offering ${task.id} (my ETA ${eta.toFixed(1)} s) to faster idle peers.`);
  }

  /** Holder's own cost to its pickup, in the same units as bidCost. */
  _holderCost(pickup) {
    const ls = this.ls;
    const risk = ls.riskScore > 0 ? this.agent.raceEvaluator.bidWeights.w_r * ls.riskScore * 120.0 : 0;
    const route = this._route(pickup);
    const eta = route.len / CRUISE_PX_S;
    return eta + (risk - 0.5 * ls.battery) / CRUISE_PX_S;
  }

  /** Board side of a handover: the lease moves, creation / award / start times stay. */
  recordHandover(taskRegistry, taskId, path) {
    const t = taskRegistry.getTaskById(taskId);
    if (!t) return;
    const startedSim = t.startedSim, startTime = t.startTime;
    taskRegistry.reassignTask(taskId, this.agent.robotId, path);
    taskRegistry.startTask(taskId);
    t.startedSim = startedSim;
    t.startTime = startTime;
    t.handovers = (t.handovers || 0) + 1;
  }

  // ── bid cost ─────────────────────────────────────────────────────────────

  /** Local bid cost for a task (lower is better). */
  bidCost(task) {
    const ls = this.ls;
    const risk = ls.riskScore > 0 ? this.agent.raceEvaluator.bidWeights.w_r * ls.riskScore * 120.0 : 0;
    if (!aceFeature("ws3-eta-bid")) {
      // Legacy Contract-Net cost (px): straight line - battery + risk.
      return Math.hypot(task.pickup.x - ls.x, task.pickup.y - ls.y) - 0.5 * ls.battery + risk;
    }
    let eta = 0, from = null;
    if (ls.currentTaskId) {
      // Busy bidder (ws3-busy-bid): remaining time, then the leg from my drop.
      const rem = this._busyRemaining();
      if (rem === null) return Infinity;
      eta = rem;
      from = ls.currentGoal;
    }
    const route = this._route(task.pickup, from);
    eta += route.len / CRUISE_PX_S;
    // Legacy weights converted from px to seconds at cruise speed.
    return eta + (risk - 0.5 * ls.battery) / CRUISE_PX_S;
  }

  /** My own planned route to a point (from here or `from`) and its length (cached per endpoints). */
  _route(pt, from = null) {
    const ls = this.ls;
    const a = from || ls;
    const key = `${Math.round(a.x)},${Math.round(a.y)}>${Math.round(pt.x)},${Math.round(pt.y)}`;
    let r = this._pathCache.get(key);
    if (r === undefined) {
      const path = this.agent.localPlanner.planPath({ x: a.x, y: a.y }, { x: pt.x, y: pt.y });
      let len = 0;
      for (let i = 1; i < path.length; i++) len += Math.hypot(path[i].x - path[i - 1].x, path[i].y - path[i - 1].y);
      r = { path, len };
      if (this._pathCache.size > 256) this._pathCache.clear();
      this._pathCache.set(key, r);
    }
    return r;
  }


}
